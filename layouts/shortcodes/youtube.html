{{- /* Overrides Hugo's built-in youtube shortcode to support a `start` timestamp (seconds). Usage: {{< youtube ID >}} or {{< youtube id="ID" start="630" >}} */ -}} {{- $id := .Get "id" | default (.Get 0) -}} {{- $src := printf "https://www.youtube.com/embed/%s" $id -}} {{- with .Get "start" }}{{ $src = printf "%s?start=%s" $src . }}{{ end -}}
